# recall — secret redaction patterns (one Python regex per line, '#' comments ignored).
# Any capture groups in a pattern are replaced with «redacted»; a pattern with no groups has
# its whole match redacted. These run over BOTH the command log and the output transcript
# before anything is written to a session .md. Add your own patterns as needed.

# AWS
\b(AKIA[0-9A-Z]{16})\b
(?i)\baws_secret_access_key\b\s*[=:]\s*("[^"]*"|'[^']*'|\S+)

# Generic secret-named key/value assignments (KEY=..., KEY: ...)
(?i)\b(?:password|passwd|pwd|secret|token|api[_-]?key|access[_-]?key|private[_-]?key|client[_-]?secret|auth[_-]?token)\b\s*[=:]\s*("[^"]*"|'[^']*'|\S+)

# export SECRET=...
(?i)\bexport\s+\w*(?:password|secret|token|key)\w*=("[^"]*"|'[^']*'|\S+)

# Bearer tokens / Authorization headers
(?i)\b(?:bearer|authorization:?)\s+([A-Za-z0-9._~+/-]{12,}=*)

# JSON Web Tokens
\b(eyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,})\b

# Provider tokens (GitHub, Slack, OpenAI/Anthropic-style sk-)
\b(gh[pousr]_[A-Za-z0-9]{20,})\b
\b(xox[baprs]-[A-Za-z0-9-]{10,})\b
\b(sk-[A-Za-z0-9]{20,})\b

# Private key PEM headers (a remnant on one line)
-----BEGIN[^-]*PRIVATE KEY-----
